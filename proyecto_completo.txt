import os

# Carpetas y extensiones que sí queremos incluir
EXCLUIR_DIRS = {'venv', '__pycache__', '.git', 'migrations'}
EXTENSIONES = {'.py', '.html', '.json', '.txt'}

with open('proyecto_completo.txt', 'w', encoding='utf-8') as salida:
    for raiz, dirs, archivos in os.walk('.'):
        # Filtrar carpetas ignoradas
        dirs[:] = [d for d in dirs if d not in EXCLUIR_DIRS]
        for archivo in sorted(archivos):
            if archivo == 'exportar.py' or archivo == 'proyecto_completo.txt':
                continue
            if any(archivo.endswith(ext) for ext in EXTENSIONES):
                ruta = os.path.join(raiz, archivo)
                salida.write(f"\n{'='*60}\nARCHIVO: {ruta}\n{'='*60}\n")
                try:
                    with open(ruta, 'r', encoding='utf-8') as f:
                        salida.write(f.read() + "\n")
                except Exception as e:
                    salida.write(f"[Error leyendo archivo: {e}]\n")

print("Listo: se generó 'proyecto_completo.txt'")

#!/usr/bin/env python
"""Django's command-line utility for administrative tasks."""
import os
import sys


def main():
    """Run administrative tasks."""
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'blog_django.settings')
    try:
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        raise ImportError(
            "Couldn't import Django. Are you sure it's installed and "
            "available on your PYTHONPATH environment variable? Did you "
            "forget to activate a virtual environment?"
        ) from exc
    execute_from_command_line(sys.argv)


if __name__ == '__main__':
    main()

from django import forms
from django.contrib.auth import get_user_model
from django.contrib.auth.forms import UserCreationForm

User = get_user_model()

class RegisterForm(UserCreationForm):
    email = forms.EmailField(required=True, label="Correo electronico")

    class Meta(UserCreationForm.Meta):
        model = User
        fields = ("username", "email")

    def clean_email(self):
        email = self.cleaned_data["email"].lower()
        if User.objects.filter(email__iexact=email).exists():
            raise forms.ValidationError("Ya existe un usuario con ese correo.")
        return email

from django.shortcuts import render, redirect
from django.contrib import messages
from django.contrib.auth import login

from .forms import RegisterForm

# Create your views here.
def register(request):
    if request.user.is_authenticated:
        return redirect("blogs_django:posts")
    if request.method == "POST":
        form = RegisterForm(request.POST)
        if form.is_valid():
            user = form.save()
            login(request, user)
            messages.success(request, f"¡Bienvenido, {user.username}!")
            return redirect("blogs_django:posts")
    else:
        form = RegisterForm()
    return render(request, "accounts/register.html", {"form": form})

from django import forms
from .models import Post, Tag

INPUT_CLASSES = (
    'w-full px-3 py-2 bg-gray-950 border border-gray-700 rounded-lg '
    'text-gray-100 focus:outline-none focus:border-indigo-500'

class TagForm(forms.ModelForm):
    class Meta:
        model = Tag
        fields = ['name']
        labels = {
            'name': 'Nombre',
        }
        widgets = {
            'name': forms.TextInput(attrs={
                'class': INPUT_CLASSES,
                'placeholder': 'Ej: Python, Django, Frontend...',
            }),
        }

class PostForm(forms.ModelForm):
    class Meta:
        model = Post
        fields = ['title', 'content', 'type', 'tags']
        labels = {
            'title': 'Título',
            'content': 'Contenido',
            'type': 'Estado',
            'tags': 'Tags (puedes seleccionar varios)',
        }
        widgets = {
            'title': forms.TextInput(attrs={
                'class': INPUT_CLASSES,
                'placeholder': 'Título del post',

                'class': INPUT_CLASSES,
                'rows': 5,

            }),
            'type': forms.Select(attrs={
                'class': INPUT_CLASSES,
            }),
            'tags': forms.SelectMultiple(attrs={
                'class': f'{INPUT_CLASSES} h-28',
            }),
        }

from django.db import models
from django.contrib.auth.models import User # Recomendado para el campo author
from django.conf import settings

class Tag(models.Model):
	"""Representa una etiqueta para categorizar los posts."""
	name = models.CharField(max_length=50, unique=True)

	def __str__(self):
		return self.name

class Post(models.Model):
	POST_TYPES = [
		("borrador", "Borrador"),
		("publicado", "Publicado"),
		("archivado", "Archivado"),
	]
	"""Representa un posteo o entrada dentro de la plataforma."""
	title = models.CharField(max_length=200)
	author = models.ForeignKey(User, on_delete=models.CASCADE, related_name='posts')
	content = models.TextField()
	likes = models.PositiveIntegerField(default=0)
	tags = models.ManyToManyField(Tag, related_name='posts')
	created_at = models.DateTimeField(auto_now_add=True)
	modified_at = models.DateTimeField(auto_now=True)
	# Opciones para el campo 'type' (estado del post)
	type = models.CharField(
		max_length=20,
		choices=POST_TYPES,
		default='borrador',
	)
	def __str__(self):
		return f"{self.title} - {self.author} ({self.created_at})"

from django.shortcuts import render, redirect, get_object_or_404
from django.contrib.auth.decorators import login_required
from .models import Post, Tag
from .forms import PostForm, TagForm

def index(request):
	return render(request, 'blogs_django/index.html')

def tags(request):
	tags = Tag.objects.all()
	context = {'tags': tags}
	return render(request, 'blogs_django/tags.html', context)

def tag(request, tag_id):
	tag = Tag.objects.get(id=tag_id)
	posts = tag.posts.order_by('created_at')
	context = {'tag': tag, 'posts': posts}
	return render(request, 'blogs_django/tag.html', context)

def create_tag(request):
    if request.method == 'POST':
        form = TagForm(request.POST)
        if form.is_valid():
            form.save()
            return redirect('blogs_django:tags')
    else:
        form = TagForm()
    return render(request, 'blogs_django/create_tag.html', {'form': form})

def posts(request):
	posts = Post.objects.all().order_by('created_at')
	context = {'posts': posts}
	return render(request, 'blogs_django/posts.html', context)

def post(request, post_id):
	post = Post.objects.get(id=post_id)
	context = {'post': post}
	return render(request, 'blogs_django/post.html', context)

@login_required
def create_post(request):
    if request.method == 'POST':
        form = PostForm(request.POST)
        if form.is_valid():
            post = form.save(commit=False)
            post.author = request.user
            post.save()
            form.save_m2m()
            return redirect('blogs_django:post', post.id)
    else:
        form = PostForm()
    return render(request, 'blogs_django/create_post.html', {'form': form})

@login_required
def edit_post(request, post_id):
    post = get_object_or_404(Post, id=post_id)
    if request.method == 'POST':
        form = PostForm(request.POST, instance=post)
        if form.is_valid():
            form.save()
            return redirect('blogs_django:post', post.id)
    else:
        form = PostForm(instance=post)
    context = {'form': form, 'post': post}
    return render(request, 'blogs_django/edit_post.html', context)

@login_required
def delete_post(request, post_id):
    post = get_object_or_404(Post, id=post_id)
    if request.method == 'POST':
        post.delete()
        return redirect('blogs_django:posts')
    return redirect('blogs_django:post', post.id)
